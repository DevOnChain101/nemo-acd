async function signIn(email, password) {
  const response = await fetch(
    '/api/auth/login', {
      method: 'POST',
      body: JSON.stringify({ email, password })
    }
  );

  if (!response.ok) {
    throw new Error('Auth failed');
  }
  return await response.json();
}
